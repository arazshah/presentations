🎯 GeoPandas چیست؟

GeoPandas یک کتابخانه متن‌باز پایتون است که قابلیت‌های کار با داده‌های جغرافیایی (Geospatial) را به Pandas اضافه می‌کند. این کتابخانه امکان خواندن، نوشتن، تحلیل و تجسم داده‌های مکانی را با سادگی Pandas فراهم می‌کند.

from shapely.geometry import Point, Polygon
import matplotlib.pyplot as plt

# ═══════════════════════════════════════════════════════════
# ایجاد GeoDataFrame از نقاط
# ═══════════════════════════════════════════════════════════

# داده‌های شهرهای ایران
cities_data = {
    'city': ['تهران', 'اصفهان', 'شیراز', 'تبریز', 'مشهد'],
    'population': [8700000, 2000000, 1600000, 1600000, 3000000],
    'latitude': [35.6892, 32.6546, 29.5918, 38.0800, 36.2605],
    'longitude': [51.3890, 51.6680, 52.5837, 46.2919, 59.6168]
}

# ایجاد هندسه از مختصات
geometry = [Point(lon, lat) for lon, lat in 
            zip(cities_data['longitude'], cities_data['latitude'])]

# ایجاد GeoDataFrame
gdf = gpd.GeoDataFrame(
    cities_data, 
    geometry=geometry,
    crs="EPSG:4326"  # WGS84
)

print(gdf)
print(f"\nنوع: {type(gdf)}")
print(f"\nCRS: {gdf.crs}")


# ═══════════════════════════════════════════════════════════
# نمایش روی نقشه
# ═══════════════════════════════════════════════════════════

fig, ax = plt.subplots(figsize=(10, 8))

    'name': ['تهران', 'اصفهان', 'شیراز'],

    ]

PY
crs_management.py 
import geopandas as gpd
from shapely.geometry import Point

# ═══════════════════════════════════════════════════════════
# سیستم‌های مختصات رایج
# ═══════════════════════════════════════════════════════════

"""
EPSG:4326  - WGS84 (GPS) - جغرافیایی (درجه)
EPSG:3857  - Web Mercator - برای نقشه‌های وب
EPSG:32639 - UTM Zone 39N - برای ایران (متر)
EPSG:32640 - UTM Zone 40N - برای شرق ایران (متر)
"""


# ═══════════════════════════════════════════════════════════
# تنظیم CRS
# ═══════════════════════════════════════════════════════════

# هنگام ایجاد
gdf = gpd.GeoDataFrame(
    {'name': ['تهران']},
    geometry=[Point(51.3890, 35.6892)],
    crs="EPSG:4326"
)

# یا بعداً (فقط تنظیم، بدون تبدیل)
gdf = gdf.set_crs("EPSG:4326")

# اگر CRS ندارد و می‌خواهید تنظیم کنید
gdf = gdf.set_crs("EPSG:4326", allow_override=True)


# ═══════════════════════════════════════════════════════════
# تبدیل CRS (Reprojection)
# ═══════════════════════════════════════════════════════════

# تبدیل به UTM برای محاسبات دقیق
gdf_utm = gdf.to_crs("EPSG:32639")

print(f"CRS اصلی: {gdf.crs}")
print(f"CRS جدید: {gdf_utm.crs}")

# مختصات قبل و بعد
print(f"\nمختصات WGS84: {gdf.geometry[0]}")
print(f"مختصات UTM: {gdf_utm.geometry[0]}")


# ═══════════════════════════════════════════════════════════
# اهمیت CRS در محاسبات
# ═══════════════════════════════════════════════════════════

# ایجاد چندضلعی
from shapely.geometry import Polygon

polygon = Polygon([
    (51.0, 35.0),
    (52.0, 35.0),
    (52.0, 36.0),
    (51.0, 36.0)
])

gdf = gpd.GeoDataFrame(geometry=[polygon], crs="EPSG:4326")

# مساحت در WGS84 (درجه مربع - بی‌معنی!)
print(f"مساحت در WGS84: {gdf.area[0]:.6f} درجه مربع")

# تبدیل به UTM برای مساحت دقیق
gdf_utm = gdf.to_crs("EPSG:32639")
area_km2 = gdf_utm.area[0] / 1_000_000
print(f"مساحت در UTM: {area_km2:.2f} کیلومتر مربع")


# ═══════════════════════════════════════════════════════════
# بررسی و دریافت اطلاعات CRS
# ═══════════════════════════════════════════════════════════

crs = gdf.crs

print(f"نام: {crs.name}")
print(f"کد EPSG: {crs.to_epsg()}")
print(f"واحد: {crs.axis_info[0].unit_name}")
print(f"جغرافیایی است: {crs.is_geographic}")
print(f"تصویری است: {crs.is_projected}")

# WKT کامل
print(f"\nWKT:\n{crs.to_wkt(pretty=True)}")
🌐

انتخاب CRS مناسب برای ایران

  • EPSG:4326 - برای ذخیره و تبادل داده
  • EPSG:32639 - UTM Zone 39N برای غرب و مرکز ایران
  • EPSG:32640 - UTM Zone 40N برای شرق ایران
  • EPSG:2059 - سیستم مختصات ملی ایران (Iran National Grid)

def slow_spatial_join(points, polygons):
    results = []

"""
پروژه: تحلیل مکان‌یابی بهینه فروشگاه جدید
هدف: پیدا کردن بهترین مکان برای افتتاح فروشگاه جدید
"""

import geopandas as gpd
import pandas as pd
from shapely.geometry import Point
import matplotlib.pyplot as plt
import numpy as np

class StoreLocationAnalyzer:
    """تحلیلگر مکان‌یابی فروشگاه"""
    
    def __init__(self, city_boundary, existing_stores, population_data):
        self.city = city_boundary
        self.stores = existing_stores
        self.population = population_data
        
        # تبدیل به UTM برای محاسبات دقیق
        self.crs_utm = "EPSG:32639"
        self.city_utm = self.city.to_crs(self.crs_utm)
        self.stores_utm = self.stores.to_crs(self.crs_utm)
        self.population_utm = self.population.to_crs(self.crs_utm)
    
    def calculate_store_coverage(self, radius_m=1000):
        """محاسبه پوشش فروشگاه‌های موجود"""
        
        # Buffer فروشگاه‌ها
        self.stores_utm['coverage'] = self.stores_utm.buffer(radius_m)
        
        # Union همه پوشش‌ها
        total_coverage = self.stores_utm['coverage'].unary_union
        
        # محاسبه درصد پوشش
        city_area = self.city_utm.area.sum()
        coverage_area = total_coverage.area
        coverage_percent = (coverage_area / city_area) * 100
        
        print(f"درصد پوشش فعلی: {coverage_percent:.1f}%")
        
        return total_coverage
    
    def find_underserved_areas(self, radius_m=1000):
        """پیدا کردن مناطق بدون پوشش"""
        
        coverage = self.calculate_store_coverage(radius_m)
        city_polygon = self.city_utm.unary_union
        
        # مناطق بدون پوشش
        underserved = city_polygon.difference(coverage)
        
        return gpd.GeoDataFrame(geometry=[underserved], crs=self.crs_utm)
    
    def calculate_demand_score(self, grid_size=500):
        """محاسبه امتیاز تقاضا برای هر سلول"""
        
        # ایجاد گرید
        bounds = self.city_utm.total_bounds
        x_coords = np.arange(bounds[0], bounds[2], grid_size)
        y_coords = np.arange(bounds[1], bounds[3], grid_size)
        
        grid_cells = []
        for x in x_coords:
            for y in y_coords:
                cell = Point(x + grid_size/2, y + grid_size/2).buffer(grid_size/2, cap_style=3)
                grid_cells.append(cell)
        
        grid = gpd.GeoDataFrame(geometry=grid_cells, crs=self.crs_utm)
        
        # فیلتر سلول‌های داخل شهر
        grid = grid[grid.intersects(self.city_utm.unary_union)]
        
        # محاسبه جمعیت هر سلول
        joined = gpd.sjoin(self.population_utm, grid, predicate='within')
        pop_per_cell = joined.groupby('index_right')['population'].sum()
        grid['population'] = grid.index.map(pop_per_cell).fillna(0)
        
        # محاسبه فاصله از نزدیک‌ترین فروشگاه
        grid['centroid'] = grid.geometry.centroid
        stores_union = self.stores_utm.unary_union
        grid['distance_to_store'] = grid['centroid'].distance(stores_union)
        
        # محاسبه امتیاز (جمعیت بالا + فاصله زیاد = امتیاز بالا)
        grid['demand_score'] = (
            grid['population'] / grid['population'].max() * 0.6 +
            grid['distance_to_store'] / grid['distance_to_store'].max() * 0.4
        )
        
        return grid
    
    def find_best_locations(self, n=5, grid_size=500):
        """پیدا کردن بهترین مکان‌ها"""
        
        grid = self.calculate_demand_score(grid_size)
        
        # مرتب‌سازی بر اساس امتیاز
        best = grid.nlargest(n, 'demand_score')
        
        # تبدیل به نقطه
        best['geometry'] = best['centroid']
        
        return best[['geometry', 'population', 'distance_to_store', 'demand_score']]
    
    def visualize_analysis(self):
        """نمایش نتایج تحلیل"""
        
        fig, axes = plt.subplots(2, 2, figsize=(16, 14))
        
        # نقشه 1: فروشگاه‌های موجود و پوشش
        ax1 = axes[0, 0]
        self.city_utm.plot(ax=ax1, color='lightgray', edgecolor='black')
        coverage = self.stores_utm.buffer(1000)
        gpd.GeoSeries(coverage).plot(ax=ax1, alpha=0.3, color='blue')
        self.stores_utm.plot(ax=ax1, color='red', markersize=100, marker='*')
        ax1.set_title('فروشگاه‌های موجود و پوشش 1 کیلومتری')
        
        # نقشه 2: تراکم جمعیت
        ax2 = axes[0, 1]
        self.population_utm.plot(ax=ax2, column='population', cmap='YlOrRd', legend=True)
        self.city_utm.boundary.plot(ax=ax2, color='black')
        ax2.set_title('تراکم جمعیت')
        
        # نقشه 3: امتیاز تقاضا
        ax3 = axes[1, 0]
        grid = self.calculate_demand_score()
        grid.plot(ax=ax3, column='demand_score', cmap='RdYlGn', legend=True)
        self.stores_utm.plot(ax=ax3, color='black', markersize=50)
        ax3.set_title('امتیاز تقاضا')
        
        # نقشه 4: بهترین مکان‌ها
        ax4 = axes[1, 1]
        self.city_utm.plot(ax=ax4, color='lightgray', edgecolor='black')
        self.stores_utm.plot(ax=ax4, color='blue', markersize=100, marker='o', label='موجود')
        best = self.find_best_locations(5)
        best.plot(ax=ax4, color='green', markersize=200, marker='*', label='پیشنهادی')
        ax4.legend()
        ax4.set_title('بهترین مکان‌های پیشنهادی')
        
        for ax in axes.flat:
            ax.axis('off')
        
        plt.tight_layout()
        plt.savefig('store_location_analysis.png', dpi=300, bbox_inches='tight')
        plt.show()

"""
پروژه: تحلیل ریسک سیلاب
هدف: شناسایی مناطق در معرض خطر و تخمین خسارت
"""

import geopandas as gpd
import pandas as pd
import numpy as np
from shapely.geometry import Point
import matplotlib.pyplot as plt
from rasterstats import zonal_stats

class FloodRiskAnalyzer:
    """تحلیلگر ریسک سیلاب"""
    
    def __init__(self, dem_path, rivers, buildings, flood_zone):
        self.dem_path = dem_path
        self.rivers = rivers
        self.buildings = buildings
        self.flood_zone = flood_zone
        
        # CRS یکسان
        self.crs = self.buildings.crs
        self.rivers = self.rivers.to_crs(self.crs)
        self.flood_zone = self.flood_zone.to_crs(self.crs)
    
    def calculate_elevation_stats(self):
        """محاسبه آمار ارتفاعی ساختمان‌ها"""
        
        stats = zonal_stats(
            self.buildings,
            self.dem_path,
            stats=['min', 'mean']
        )
        
        self.buildings['elevation_min'] = [s['min'] for s in stats]
        self.buildings['elevation_mean'] = [s['mean'] for s in stats]
        
        return self.buildings
    
    def calculate_distance_to_river(self):
        """محاسبه فاصله تا رودخانه"""
        
        rivers_union = self.rivers.unary_union
        self.buildings['distance_to_river'] = self.buildings.geometry.centroid.distance(rivers_union)
        
        return self.buildings
    
    def identify_at_risk_buildings(self):
        """شناسایی ساختمان‌های در معرض خطر"""
        
        # ساختمان‌های داخل منطقه سیل‌گیر
        flood_polygon = self.flood_zone.unary_union
        self.buildings['in_flood_zone'] = self.buildings.intersects(flood_polygon)
        
        # محاسبه ارتفاع و فاصله
        self.calculate_elevation_stats()
        self.calculate_distance_to_river()
        
        # محاسبه امتیاز ریسک
        # فاکتورها: در منطقه سیل‌گیر، ارتفاع پایین، نزدیک به رودخانه
        self.buildings['risk_score'] = 0.0
        
        # در منطقه سیل‌گیر
        self.buildings.loc[self.buildings['in_flood_zone'], 'risk_score'] += 40
        
        # ارتفاع پایین (نرمال‌سازی معکوس)
        elev_normalized = 1 - (self.buildings['elevation_min'] - self.buildings['elevation_min'].min()) / \
                              (self.buildings['elevation_min'].max() - self.buildings['elevation_min'].min())
        self.buildings['risk_score'] += elev_normalized * 30
        
        # فاصله کم از رودخانه
        dist_normalized = 1 - (self.buildings['distance_to_river'] - self.buildings['distance_to_river'].min()) / \
                              (self.buildings['distance_to_river'].max() - self.buildings['distance_to_river'].min())
        self.buildings['risk_score'] += dist_normalized * 30
        
        # طبقه‌بندی ریسک
        self.buildings['risk_level'] = pd.cut(
            self.buildings['risk_score'],
            bins=[0, 30, 50, 70, 100],
            labels=['کم', 'متوسط', 'بالا', 'بسیار بالا']
        )
        
        return self.buildings
    
    def estimate_damage(self, price_per_sqm=50000000):
        """تخمین خسارت احتمالی"""
        
        at_risk = self.buildings[self.buildings['in_flood_zone']].copy()
        
        # محاسبه مساحت
        at_risk['area_sqm'] = at_risk.geometry.area
        
        # تخمین خسارت (فرض: 30% ارزش ملک)
        at_risk['estimated_damage'] = at_risk['area_sqm'] * price_per_sqm * 0.3
        
        total_damage = at_risk['estimated_damage'].sum()
        
        print(f"تعداد ساختمان در معرض خطر: {len(at_risk)}")
        print(f"کل خسارت تخمینی: {total_damage/1e9:.2f} میلیارد تومان")
        
        return at_risk
    
    def generate_report(self):
        """تولید گزارش"""
        
        self.identify_at_risk_buildings()
        
        report = {
            'total_buildings': len(self.buildings),
            'buildings_in_flood_zone': self.buildings['in_flood_zone'].sum(),
            'risk_distribution': self.buildings['risk_level'].value_counts().to_dict(),
            'average_risk_score': self.buildings['risk_score'].mean()
        }
        
        return report
    
    def visualize(self):
        """نمایش نتایج"""
        
        fig, axes = plt.subplots(2, 2, figsize=(16, 14))
        
        # نقشه 1: منطقه سیل‌گیر و رودخانه
        ax1 = axes[0, 0]
        self.flood_zone.plot(ax=ax1, color='lightblue', alpha=0.5, label='منطقه سیل‌گیر')
        self.rivers.plot(ax=ax1, color='blue', linewidth=2, label='رودخانه')
        self.buildings.plot(ax=ax1, color='gray', alpha=0.5)
        ax1.set_title('منطقه سیل‌گیر و رودخانه')
        ax1.legend()
        
        # نقشه 2: ساختمان‌های در معرض خطر
        ax2 = axes[0, 1]
        self.buildings[~self.buildings['in_flood_zone']].plot(ax=ax2, color='green', alpha=0.5, label='امن')
        self.buildings[self.buildings['in_flood_zone']].plot(ax=ax2, color='red', alpha=0.7, label='در معرض خطر')
        ax2.set_title('ساختمان‌های در معرض خطر')
        ax2.legend()
        
        # نقشه 3: امتیاز ریسک
        ax3 = axes[1, 0]
        self.buildings.plot(ax=ax3, column='risk_score', cmap='RdYlGn_r', legend=True)
        ax3.set_title('امتیاز ریسک')
        
        # نقشه 4: سطح ریسک
        ax4 = axes[1, 1]
        colors = {'کم': 'green', 'متوسط': 'yellow', 'بالا': 'orange', 'بسیار بالا': 'red'}
        for level, color in colors.items():
            subset = self.buildings[self.buildings['risk_level'] == level]
            if len(subset) > 0:
                subset.plot(ax=ax4, color=color, label=level, alpha=0.7)
        ax4.set_title('سطح ریسک')
        ax4.legend()
        
        for ax in axes.flat:
            ax.axis('off')
        
        plt.tight_layout()
        plt.savefig('flood_risk_analysis.png', dpi=300, bbox_inches='tight')
        plt.show()
