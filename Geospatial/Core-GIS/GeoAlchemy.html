from database import Base

# ═══════════════════════════════════════════════════════════
# مدل با Spatial Index
# ═══════════════════════════════════════════════════════════

class Store(Base):
    """فروشگاه با ایندکس مکانی"""
    __tablename__ = 'stores'
    
    id = Column(Integer, primary_key=True)
    name = Column(String(100), nullable=False)
    address = Column(String(200))
    store_type = Column(String(50))
    is_active = Column(Boolean, default=True)
    
    # موقعیت فروشگاه
    location = Column(
        Geometry(
            geometry_type='POINT',
            srid=4326,
            spatial_index=True  # ایجاد خودکار GiST index
        ),
        nullable=False
    )
    
    # ایندکس‌های اضافی
    __table_args__ = (
        Index('idx_store_name', 'name'),
        Index('idx_store_type_active', 'store_type', 'is_active'),
    )
    
    @property
    def latitude(self):
        """استخراج عرض جغرافیایی"""
        if self.location:
            point = to_shape(self.location)
            return point.y
        return None
    
    @property
    def longitude(self):
        """استخراج طول جغرافیایی"""
        if self.location:
            point = to_shape(self.location)
            return point.x
        return None
    
    @classmethod
    def create_from_coords(cls, name, lat, lon, **kwargs):
        """ایجاد فروشگاه از مختصات"""
        point = Point(lon, lat)
        return cls(
            name=name,
            location=from_shape(point, srid=4326),
            **kwargs
        )
    
    def to_geojson(self):
        """تبدیل به GeoJSON Feature"""
        if self.location:
            point = to_shape(self.location)
            return {
                "type": "Feature",
                "geometry": mapping(point),
                "properties": {
                    "id": self.id,
                    "name": self.name,
                    "address": self.address,
                    "store_type": self.store_type
                }
            }
        return None

# Event listener برای validation
@event.listens_for(ValidatedPolygon, 'before_insert')
@event.listens_for(ValidatedPolygon, 'before_update')
def validate_geometry(mapper, connection, target):
    """اعتبارسنجی geometry قبل از ذخیره"""
    if target.geom:
        shape = to_shape(target.geom)
        if not shape.is_valid:
            # تلاش برای تعمیر
            fixed = shape.buffer(0)
            target.geom = from_shape(fixed, srid=4326)

"""
تفاوت‌های کلیدی:

1. واحد اندازه‌گیری:
   - Geometry: واحد SRID (درجه برای 4326)
   - Geography: همیشه متر

2. دقت:
   - Geometry: برای مناطق کوچک دقیق‌تر
   - Geography: برای فواصل بزرگ دقیق‌تر

3. سرعت:
   - Geometry: سریع‌تر
   - Geography: کندتر (محاسبات پیچیده‌تر)

4. توابع:
   - Geometry: همه توابع ST_*

def find_stores_within_radius(
    session: Session,
    lat: float,
    lon: float,

        select(Store).where(Store.id.in_(store_ids))

                distance = session.execute(
                    select(func.ST_DistanceSphere(s1.location, s2.location))
                ).scalar()
                matrix[s1.id][s2.id] = distance
            else:
                matrix[s1.id][s2.id] = 0
    
    return matrix

def get_elevation_at_point(

def get_elevation_profile(
    session: Session,
    line_wkt: str,
    num_points: int = 100
) -> list:
    """پروفایل ارتفاعی در طول یک خط"""
    
    # ایجاد نقاط روی خط
    query = text("""
        WITH line AS (
            SELECT ST_GeomFromText(:line_wkt, 4326) as geom
        ),
        points AS (
            SELECT 
                generate_series(0, :num_points) as idx,
                ST_LineInterpolatePoint(
                    (SELECT geom FROM line),
                    generate_series(0, :num_points)::float / :num_points
                ) as point
        )
        SELECT 
            p.idx,
            ST_X(p.point) as lon,
            ST_Y(p.point) as lat,
            ST_Value(e.rast, p.point) as elevation
        FROM points p
        LEFT JOIN elevation_tiles e ON ST_Intersects(e.rast, p.point)
        ORDER BY p.idx
    """)
    
    result = session.execute(
        query,
        {'line_wkt': line_wkt, 'num_points': num_points}
    )
    
    return [
        {'idx': r.idx, 'lon': r.lon, 'lat': r.lat, 'elevation': r.elevation}
        for r in result
    ]

        func.ST_SummaryStats(

    total_sum = sum(r[0][1] for r in results if r[0])
    all_mins = [r[0][4] for r in results if r[0]]
    all_maxs = [r[0][5] for r in results if r[0]]
    
    return {
        'count': total_count,
        'mean': total_sum / total_count if total_count > 0 else None,
        'min': min(all_mins) if all_mins else None,
        'max': max(all_maxs) if all_maxs else None

def raster_to_contours(

            (ST_DumpAsPolygons(
                ST_Reclass(
                    rast,
                    1,
                    CONCAT('[0-', :interval, '):1, [', :interval, '-', :interval*2, '):2'),
                    '8BUI',
                    0
                )
            )).*
        FROM elevation_tiles
        WHERE id = :tile_id
    """)

        SELECT 
            level,
            ST_AsGeoJSON(geom) as geojson
        FROM ST_Contour(

            1,  -- band
            :interval,  -- interval
            0   -- start value
        )
    """)

def optimized_radius_search(
    session: Session,
    lat: float,
    lon: float,
    radius_m: float
):
    """جستجوی شعاعی بهینه"""
    
    # استفاده از && برای فیلتر اولیه با ایندکس
    # سپس ST_DWithin برای فیلتر دقیق
    
    query = text("""
        SELECT *
        FROM stores
        WHERE location && ST_Expand(
            ST_SetSRID(ST_MakePoint(:lon, :lat), 4326)::geography::geometry,
            :radius / 111000  -- تبدیل متر به درجه (تقریبی)
        )
        AND ST_DWithin(
            location::geography,
            ST_SetSRID(ST_MakePoint(:lon, :lat), 4326)::geography,
            :radius
        )
    """)
    
    return session.execute(
        query,
        {'lat': lat, 'lon': lon, 'radius': radius_m}
    ).all()

    while offset < total:
        # بروزرسانی یک دسته
        session.execute(text("""
            UPDATE stores s
            SET district_id = d.id
            FROM districts d
            WHERE ST_Within(s.location, d.boundary)
            AND s.district_id IS NULL
            AND s.id IN (
                SELECT id FROM stores 
                WHERE district_id IS NULL 
                LIMIT :batch_size
            )
        """), {'batch_size': batch_size})
        
        session.commit()
        offset += batch_size
        print(f"Processed {min(offset, total)}/{total}")

@app.get("/stores/nearby/", response_model=List[StoreResponse], tags=["Spatial"])
async def find_nearby_stores(
    lat: float = Query(..., ge=-90, le=90, description="عرض جغرافیایی"),
    lon: float = Query(..., ge=-180, le=180, description="طول جغرافیایی"),
    radius_km: float = Query(default=5, ge=0.1, le=100, description="شعاع جستجو (کیلومتر)"),
    category: Optional[str] = Query(default=None, description="دسته‌بندی"),
    limit: int = Query(default=20, ge=1, le=100),
    db: AsyncSession = Depends(get_db)
):
    """جستجوی فروشگاه‌های نزدیک"""
    
    point = func.ST_SetSRID(func.ST_MakePoint(lon, lat), 4326)
    radius_m = radius_km * 1000
    
    # فاصله
    distance_expr = func.ST_Distance(
        Store.location,
        func.ST_GeogFromText(f'POINT({lon} {lat})')
    )
    
    query = (
        select(Store, distance_expr.label('distance'))
        .where(
            and_(
                func.ST_DWithin(
                    Store.location,
                    func.ST_GeogFromText(f'POINT({lon} {lat})'),
                    radius_m
                ),
                Store.is_active == True
            )
        )
        .order_by(distance_expr)
        .limit(limit)
    )
    
    if category:
        query = query.where(Store.category == category)
    
    result = await db.execute(query)
    
    stores = []
    for row in result:
        store = row.Store
        point = to_shape(store.location)
        stores.append(StoreResponse(
            id=store.id,
            name=store.name,
            address=store.address,
            phone=store.phone,
            category=store.category,
            rating=store.rating,
            is_active=store.is_active,
            latitude=point.y,
            longitude=point.x,
            distance_m=row.distance
        ))
    
    return stores

def find_nearest_node(session: Session, lat: float, lon: float) -> RoadNode:
    """پیدا کردن نزدیک‌ترین گره"""
    
    point = func.ST_SetSRID(func.ST_MakePoint(lon, lat), 4326)
    
    query = (
        select(RoadNode)
        .order_by(RoadNode.location.distance_centroid(point))
        .limit(1)
    )
    
    result = session.execute(query)
    return result.scalar_one_or_none()


def find_shortest_path_dijkstra(
    session: Session,
    source_id: int,
    target_id: int
) -> list:
    """مسیریابی با الگوریتم Dijkstra (نیاز به pgRouting)"""
    
    query = text("""
        SELECT 
            seq, node, edge, cost, agg_cost,
            ST_AsGeoJSON(e.geom) as geojson
        FROM pgr_dijkstra(
            'SELECT id, source_id as source, target_id as target, cost, reverse_cost 
             FROM road_edges',
            :source_id,
            :target_id,
            directed := true
        ) AS route
        LEFT JOIN road_edges e ON route.edge = e.id
        ORDER BY seq
    """)
    
    result = session.execute(query, {'source_id': source_id, 'target_id': target_id})
    
    return [
        {
            'seq': row.seq,
            'node': row.node,
            'edge': row.edge,
            'cost': row.cost,
            'agg_cost': row.agg_cost,
            'geojson': row.geojson
        }
        for row in result
    ]


def find_route_between_points(
    session: Session,
    start_lat: float,
    start_lon: float,
    end_lat: float,
    end_lon: float
) -> dict:
    """مسیریابی بین دو نقطه"""
    
    # پیدا کردن نزدیک‌ترین گره‌ها
    start_node = find_nearest_node(session, start_lat, start_lon)
    end_node = find_nearest_node(session, end_lat, end_lon)
    
    if not start_node or not end_node:
        return {'error': 'Could not find nearest nodes'}
    
    # مسیریابی
    path = find_shortest_path_dijkstra(session, start_node.id, end_node.id)
    
    if not path:
        return {'error': 'No route found'}
    
    # ساخت GeoJSON
    features = []
    total_cost = 0
    
    for step in path:
        if step['geojson']:
            features.append({
                'type': 'Feature',
                'geometry': json.loads(step['geojson']),
                'properties': {
                    'seq': step['seq'],
                    'cost': step['cost']
                }
            })
        total_cost = step['agg_cost']
    
    return {
        'type': 'FeatureCollection',
        'features': features,
        'properties': {
            'total_cost': total_cost,
            'start_node': start_node.id,
            'end_node': end_node.id
        }
    }


def calculate_isochrone(
    session: Session,
    center_lat: float,
    center_lon: float,
    time_minutes: int
) -> str:
    """محاسبه ایزوکرون (محدوده دسترسی در زمان مشخص)"""
    
    # پیدا کردن نزدیک‌ترین گره
    center_node = find_nearest_node(session, center_lat, center_lon)
    
    if not center_node:
        return None
    
    # تبدیل زمان به هزینه (فرض: هزینه = زمان به دقیقه)
    max_cost = time_minutes
    
    query = text("""
        WITH reachable AS (
            SELECT node, agg_cost
            FROM pgr_drivingDistance(
                'SELECT id, source_id as source, target_id as target, cost, reverse_cost 
                 FROM road_edges',
                :center_id,
                :max_cost,
                directed := true
            )
        ),
        node_points AS (
            SELECT n.location as geom
            FROM reachable r
            JOIN road_nodes n ON r.node = n.id
        )
        SELECT ST_AsGeoJSON(
            ST_ConcaveHull(ST_Collect(geom), 0.8)
        ) as isochrone
        FROM node_points
    """)
    
    result = session.execute(
        query,
        {'center_id': center_node.id, 'max_cost': max_cost}
    )
    
    return result.scalar()

    def kernel_density_estimation(
        self,
        bandwidth_km: float = 1.0,
        grid_resolution: int = 100
    ) -> Dict:
        """تخمین تراکم کرنل (KDE)"""
        
        # دریافت همه نقاط
        stores = self.session.execute(select(Store)).scalars().all()
        
        points = [to_shape(s.location) for s in stores]
        
        # محاسبه KDE با scipy
        from scipy import stats
        import numpy as np
        
        if not points:
            return {}
        
        x = np.array([p.x for p in points])
        y = np.array([p.y for p in points])
        
        # ایجاد grid
        xmin, xmax = x.min(), x.max()
        ymin, ymax = y.min(), y.max()
        
        xx, yy = np.mgrid[xmin:xmax:complex(grid_resolution), 
                         ymin:ymax:complex(grid_resolution)]
        
        positions = np.vstack([xx.ravel(), yy.ravel()])
        values = np.vstack([x, y])
        
        kernel = stats.gaussian_kde(values, bw_method=bandwidth_km/111)
        density = np.reshape(kernel(positions).T, xx.shape)
        
        return {
            'grid': {
                'xmin': xmin, 'xmax': xmax,
                'ymin': ymin, 'ymax': ymax,
                'resolution': grid_resolution
            },
            'density': density.tolist()
        }

    def nearest_neighbor_analysis(self) -> Dict:
        """تحلیل نزدیک‌ترین همسایه"""
        
        # میانگین فاصله تا نزدیک‌ترین همسایه
        query = text("""
            WITH nn_distances AS (
                SELECT 
                    s1.id,
                    MIN(ST_DistanceSphere(s1.location::geometry, s2.location::geometry)) as nn_distance
                FROM stores s1
                CROSS JOIN stores s2
                WHERE s1.id != s2.id
                GROUP BY s1.id
            )
            SELECT 
                COUNT(*) as n,
                AVG(nn_distance) as mean_nn_distance,
                STDDEV(nn_distance) as std_nn_distance,
                MIN(nn_distance) as min_nn_distance,
                MAX(nn_distance) as max_nn_distance
            FROM nn_distances
        """)
        
        result = self.session.execute(query).first()
        
        # محاسبه شاخص R (Clark-Evans)
        # R = observed_mean / expected_mean
        # expected_mean = 1 / (2 * sqrt(density))
        
        # محاسبه تراکم
        extent_result = self.session.execute(
            select(
                func.ST_Area(func.ST_ConvexHull(func.ST_Collect(Store.location))::func.geography)
            )
        ).scalar()
        
        n = result.n
        area = extent_result
        density = n / area if area else 0
        
        expected_mean = 1 / (2 * (density ** 0.5)) if density > 0 else 0
        observed_mean = result.mean_nn_distance
        
        r_index = observed_mean / expected_mean if expected_mean > 0 else 0
        
        # تفسیر:
        # R < 1: خوشه‌ای (clustered)
        # R = 1: تصادفی (random)
        # R > 1: پراکنده (dispersed)
        
        pattern = "clustered" if r_index < 0.9 else "dispersed" if r_index > 1.1 else "random"
        
        return {
            'count': n,
            'mean_nn_distance': observed_mean,
            'std_nn_distance': result.std_nn_distance,
            'min_nn_distance': result.min_nn_distance,
            'max_nn_distance': result.max_nn_distance,
            'r_index': r_index,
            'pattern': pattern
        }
