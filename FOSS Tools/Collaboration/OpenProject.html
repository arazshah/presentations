🎯 OpenProject چیست؟

OpenProject یک نرم‌افزار مدیریت پروژه متن‌باز و قدرتمند است که توسط شرکت آلمانی OpenProject GmbH توسعه داده شده است. این نرم‌افزار ترکیبی کامل از ابزارهای مدیریت پروژه کلاسیک و متدولوژی‌های Agile را ارائه می‌دهد و جایگزین مناسبی برای ابزارهای تجاری مانند Jira، Microsoft Project و Asana است.

چرا OpenProject؟

اگر به دنبال یک راه‌حل مدیریت پروژه هستید که:

  • داده‌هایتان روی سرور خودتان باشد (GDPR Compliant)
  • هزینه لایسنس نداشته باشد
  • قابلیت سفارشی‌سازی داشته باشد

OpenProject بهترین انتخاب است!

🐳 نصب با Docker (توصیه شده)

ساده‌ترین و سریع‌ترین روش نصب OpenProject استفاده از Docker است. این روش برای محیط‌های توسعه و تولید مناسب است.

YML
docker-compose.yml 
version: "3.8"

services:
  openproject:
    image: openproject/openproject:14
    container_name: openproject
    restart: unless-stopped
    ports:
      - "8080:80"
    environment:
      # تنظیمات پایگاه داده
      OPENPROJECT_HTTPS: "false"
      OPENPROJECT_HOST__NAME: "localhost:8080"
      OPENPROJECT_SECRET_KEY_BASE: "your-secret-key-here-min-64-chars"
      
      # تنظیمات ایمیل (اختیاری)
      EMAIL_DELIVERY_METHOD: "smtp"
      SMTP_ADDRESS: "smtp.gmail.com"
      SMTP_PORT: 587
      SMTP_DOMAIN: "gmail.com"
      SMTP_AUTHENTICATION: "plain"
      SMTP_USER_NAME: "your-email@gmail.com"
      SMTP_PASSWORD: "your-app-password"
      SMTP_ENABLE_STARTTLS_AUTO: "true"
      
      # تنظیمات زبان
      OPENPROJECT_DEFAULT__LANGUAGE: "fa"
      
    volumes:
      - openproject_data:/var/openproject/assets
      - openproject_pgdata:/var/openproject/pgdata
    
    healthcheck:
      test: ["CMD", "curl", "-f", "http://localhost:80/health_checks/default"]
      interval: 30s
      timeout: 10s
      retries: 3

volumes:
  openproject_data:
  openproject_pgdata:

🖥️ نصب روی Ubuntu/Debian

SH
install_ubuntu.sh 
#!/bin/bash
# نصب OpenProject روی Ubuntu 22.04/24.04

# ═══════════════════════════════════════════════════════════
# 1. اضافه کردن مخزن OpenProject
# ═══════════════════════════════════════════════════════════

# نصب پیش‌نیازها
sudo apt-get update
sudo apt-get install -y apt-transport-https ca-certificates curl gnupg

# اضافه کردن کلید GPG
curl -fsSL https://dl.packager.io/srv/opf/openproject/key | \
  sudo gpg --dearmor -o /etc/apt/keyrings/openproject.gpg

# اضافه کردن مخزن
echo "deb [signed-by=/etc/apt/keyrings/openproject.gpg] \
  https://dl.packager.io/srv/deb/opf/openproject/stable/14/ubuntu 22.04 main" | \
  sudo tee /etc/apt/sources.list.d/openproject.list

# ═══════════════════════════════════════════════════════════
# 2. نصب OpenProject
# ═══════════════════════════════════════════════════════════

sudo apt-get update
sudo apt-get install -y openproject

# ═══════════════════════════════════════════════════════════
# 3. پیکربندی
# ═══════════════════════════════════════════════════════════

# اجرای ویزارد پیکربندی
sudo openproject configure

# این ویزارد سوالاتی درباره موارد زیر می‌پرسد:
# - PostgreSQL (نصب خودکار یا استفاده از موجود)
# - Apache/Nginx (نصب و پیکربندی وب سرور)
# - SSL (Let's Encrypt یا گواهی سفارشی)
# - Hostname
# - Email settings

# ═══════════════════════════════════════════════════════════
# 4. مدیریت سرویس
# ═══════════════════════════════════════════════════════════

# شروع سرویس
sudo systemctl start openproject

# فعال‌سازی در بوت
sudo systemctl enable openproject

# بررسی وضعیت
sudo systemctl status openproject

# مشاهده لاگ‌ها
sudo journalctl -u openproject -f

# ═══════════════════════════════════════════════════════════
# 5. دستورات مفید
# ═══════════════════════════════════════════════════════════

# ری‌استارت
sudo openproject restart

# پیکربندی مجدد
sudo openproject reconfigure

# بروزرسانی
sudo apt-get update && sudo apt-get upgrade openproject

# بکاپ
sudo openproject backup

# بازیابی
sudo openproject restore /path/to/backup.tar.gz

kubernetes_install.sh 
# اضافه کردن Helm repository
helm repo add openproject https://charts.openproject.org
helm repo update

# نصب با تنظیمات پیش‌فرض
helm install openproject openproject/openproject

# نصب با تنظیمات سفارشی
helm install openproject openproject/openproject \
  --set persistence.enabled=true \
  --set persistence.size=10Gi \
  --set ingress.enabled=true \
  --set ingress.hostname=openproject.example.com \
  --set postgresql.enabled=true

# یا با فایل values
helm install openproject openproject/openproject -f values.yaml

🔒 پیکربندی SSL با Nginx

NG
nginx.conf 
# /etc/nginx/sites-available/openproject

upstream openproject {
    server 127.0.0.1:8080;
}

# Redirect HTTP to HTTPS
server {
    listen 80;
    server_name openproject.example.com;
    return 301 https://$server_name$request_uri;
}

# HTTPS Server
server {
    listen 443 ssl http2;
    server_name openproject.example.com;

    # SSL Configuration
    ssl_certificate /etc/letsencrypt/live/openproject.example.com/fullchain.pem;
    ssl_certificate_key /etc/letsencrypt/live/openproject.example.com/privkey.pem;
    ssl_protocols TLSv1.2 TLSv1.3;
    ssl_ciphers ECDHE-ECDSA-AES128-GCM-SHA256:ECDHE-RSA-AES128-GCM-SHA256:ECDHE-ECDSA-AES256-GCM-SHA384;
    ssl_prefer_server_ciphers off;
    ssl_session_cache shared:SSL:10m;
    ssl_session_timeout 1d;

    # Security Headers
    add_header X-Frame-Options "SAMEORIGIN" always;
    add_header X-Content-Type-Options "nosniff" always;
    add_header X-XSS-Protection "1; mode=block" always;
    add_header Strict-Transport-Security "max-age=31536000" always;

    # Logging
    access_log /var/log/nginx/openproject_access.log;
    error_log /var/log/nginx/openproject_error.log;

    # Client body size for file uploads
    client_max_body_size 100M;

    location / {
        proxy_pass http://openproject;
        proxy_http_version 1.1;
        proxy_set_header Upgrade $http_upgrade;
        proxy_set_header Connection "upgrade";
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
        proxy_set_header X-Forwarded-Host $host;
        proxy_set_header X-Forwarded-Port $server_port;
        
        proxy_connect_timeout 90;
        proxy_send_timeout 90;
        proxy_read_timeout 90;
        proxy_buffering off;
        proxy_request_buffering off;
    }
}

📦 Work Package چیست؟

Work Package واحد اصلی کار در OpenProject است. هر چیزی که باید انجام شود (تسک، باگ، فیچر، Epic و...) یک Work Package است. این مفهوم معادل Issue در Jira یا Task در Asana است.

انواع Work Package

نوع آیکون کاربرد
Task 📝 وظایف عمومی و کارهای روزمره
Bug 🐛 گزارش خطا و مشکلات نرم‌افزاری
Feature ⭐ قابلیت‌های جدید
Epic 🎯 کارهای بزرگ که شامل چندین Task می‌شوند

وضعیت‌های رایج:
• New: تازه ایجاد شده
• In Specification: در حال تعریف نیازمندی
• In Progress: در حال انجام
• On Hold: متوقف شده
• Testing: در حال تست
• Ready for Review: آماده بررسی
• Done/Closed: تکمیل شده
• Rejected: رد شده

🔗 روابط بین Work Packages

نوع رابطه توضیح مثال
Parent/Child رابطه والد-فرزند Epic شامل چند Task
Relates to ارتباط عمومی دو تسک مرتبط
Blocks مسدود کننده Task A باید قبل از Task B تمام شود
Blocked by مسدود شده توسط Task B منتظر Task A است
Duplicates تکراری این باگ قبلاً گزارش شده
Follows دنبال می‌کند شروع Task B پس از پایان Task A
Precedes قبل از Task A قبل از Task B است

📊 Boards

OpenProject دو نوع Board اصلی ارائه می‌دهد:

📋

Basic Board

بورد ساده با ستون‌های سفارشی. مناسب برای Kanban و گردش کار ساده. می‌توانید ستون‌ها را بر اساس Status، Assignee یا Version تنظیم کنید.

🎯

Action Board

بورد پیشرفته که با جابجایی کارت‌ها، فیلد مربوطه به‌روز می‌شود. مثلاً با کشیدن کارت به ستون "Done"، وضعیت تغییر می‌کند.

🎯 ایجاد Kanban Board

۱

ایجاد Board جدید

از منوی پروژه، Boards را انتخاب کنید. سپس + Board را کلیک کنید.

۲

انتخاب نوع Board

Action Board را انتخاب کنید و فیلد Status را به عنوان معیار ستون‌ها انتخاب کنید.

۳

اضافه کردن ستون‌ها

ستون‌های مورد نیاز را اضافه کنید: Backlog → To Do → In Progress → Review → Done

۴

تنظیم فیلترها

فیلترهای Board را تنظیم کنید. مثلاً فقط کارهای Sprint فعلی نمایش داده شوند.

چرخه Scrum در OpenProject
═══════════════════════════════════════════════════════════

1️⃣ Product Backlog
   └── ایجاد User Stories و Features
   └── اولویت‌بندی با Drag & Drop
   └── تخمین با Story Points (Custom Field)

2️⃣ Sprint Planning
   └── ایجاد Version جدید به نام Sprint
   └── انتخاب آیتم‌ها از Backlog
   └── اختصاص به Sprint (Version)
   └── تعیین Sprint Goal

3️⃣ Sprint Execution
   └── استفاده از Sprint Board
   └── Daily Standup با بررسی Board
   └── بروزرسانی وضعیت کارها
   └── ثبت زمان صرف شده

4️⃣ Sprint Review
   └── بررسی کارهای Done
   └── Demo به Stakeholders
   └── جمع‌آوری فیدبک

5️⃣ Sprint Retrospective
   └── ثبت در Wiki
   └── What went well?
   └── What to improve?
   └── Action items برای Sprint بعد

• Text: متن کوتاه

🐍 Python Client

PY
openproject_client.py 
"""
OpenProject Python Client
کلاینت پایتون برای کار با API اوپن‌پروجکت
"""

import requests
from typing import Optional, List, Dict, Any
from dataclasses import dataclass
from datetime import date


@dataclass
class WorkPackage:
    """مدل Work Package"""
    id: int
    subject: str
    status: str
    assignee: Optional[str] = None
    due_date: Optional[date] = None
    progress: int = 0


class OpenProjectClient:
    """کلاینت برای ارتباط با OpenProject API"""
    
    def __init__(self, base_url: str, api_key: str):
        self.base_url = base_url.rstrip('/') + '/api/v3'
        self.session = requests.Session()
        self.session.auth = ('apikey', api_key)
        self.session.headers.update({
            'Content-Type': 'application/json'
        })
    
    def get_projects(self) -> List[Dict[str, Any]]:
        """دریافت لیست پروژه‌ها"""
        response = self.session.get(f'{self.base_url}/projects')
        response.raise_for_status()
        return response.json()['_embedded']['elements']
    
    def get_project(self, project_id: int) -> Dict[str, Any]:
        """دریافت جزئیات یک پروژه"""
        response = self.session.get(f'{self.base_url}/projects/{project_id}')
        response.raise_for_status()
        return response.json()
    
    def get_work_packages(
        self, 
        project_id: int, 
        filters: Optional[List[Dict]] = None
    ) -> List[Dict[str, Any]]:
        """دریافت Work Packages یک پروژه"""
        params = {}
        if filters:
            params['filters'] = str(filters)
        
        response = self.session.get(
            f'{self.base_url}/projects/{project_id}/work_packages',
            params=params
        )
        response.raise_for_status()
        return response.json()['_embedded']['elements']
    
    def create_work_package(
        self,
        project_id: int,
        subject: str,
        type_id: int = 1,
        description: str = "",
        assignee_id: Optional[int] = None,
        priority_id: int = 2
    ) -> Dict[str, Any]:
        """ایجاد Work Package جدید"""
        data = {
            "subject": subject,
            "description": {
                "format": "markdown",
                "raw": description
            },
            "_links": {
                "type": {"href": f"/api/v3/types/{type_id}"},
                "priority": {"href": f"/api/v3/priorities/{priority_id}"}
            }
        }
        
        if assignee_id:
            data["_links"]["assignee"] = {
                "href": f"/api/v3/users/{assignee_id}"
            }
        
        response = self.session.post(
            f'{self.base_url}/projects/{project_id}/work_packages',
            json=data
        )
        response.raise_for_status()
        return response.json()
    
    def update_work_package(
        self,
        wp_id: int,
        lock_version: int,
        **kwargs
    ) -> Dict[str, Any]:
        """بروزرسانی Work Package"""
        data = {"lockVersion": lock_version}
        data.update(kwargs)
        
        response = self.session.patch(
            f'{self.base_url}/work_packages/{wp_id}',
            json=data
        )
        response.raise_for_status()
        return response.json()
    
    def log_time(
        self,
        work_package_id: int,
        hours: float,
        activity_id: int,
        spent_on: date,
        comment: str = ""
    ) -> Dict[str, Any]:
        """ثبت زمان"""
        # تبدیل ساعت به فرمت ISO 8601 duration
        hours_int = int(hours)
        minutes = int((hours - hours_int) * 60)
        duration = f"PT{hours_int}H{minutes}M"
        
        data = {
            "hours": duration,
            "spentOn": spent_on.isoformat(),
            "comment": {"raw": comment},
            "_links": {
                "workPackage": {
                    "href": f"/api/v3/work_packages/{work_package_id}"
                },
                "activity": {
                    "href": f"/api/v3/time_entries/activities/{activity_id}"
                }
            }
        }
        
        response = self.session.post(
            f'{self.base_url}/time_entries',
            json=data
        )
        response.raise_for_status()
        return response.json()


# ═══════════════════════════════════════════════════════════
# مثال استفاده
# ═══════════════════════════════════════════════════════════

if __name__ == "__main__":
    # ایجاد کلاینت
    client = OpenProjectClient(
        base_url="https://openproject.example.com",
        api_key="your-api-key"
    )
    
    # دریافت پروژه‌ها
    projects = client.get_projects()
    for project in projects:
        print(f"📁 {project['name']} (ID: {project['id']})")
    
    # دریافت Work Packages
    work_packages = client.get_work_packages(project_id=1)
    for wp in work_packages:
        print(f"📋 #{wp['id']}: {wp['subject']}")
    
    # ایجاد Work Package جدید
    new_wp = client.create_work_package(
        project_id=1,
        subject="Task created from Python",
        description="This task was created using Python API client",
        type_id=1,
        priority_id=2
    )
    print(f"✅ Created: #{new_wp['id']}")
    
    # ثبت زمان
    from datetime import date
    time_entry = client.log_time(
        work_package_id=new_wp['id'],
        hours=2.5,
        activity_id=1,
        spent_on=date.today(),
        comment="Worked on implementation"
    )
    print(f"⏱️ Logged: {time_entry['hours']}")

# OpenProject Backup & Restore Scripts

# ═══════════════════════════════════════════════════════════
# پشتیبان‌گیری (Packaged Installation)
# ═══════════════════════════════════════════════════════════

# بکاپ کامل
sudo openproject run backup

# بکاپ با مسیر سفارشی
sudo openproject run backup /path/to/backup/directory

# بکاپ شامل:
# - دیتابیس PostgreSQL
# - فایل‌های آپلود شده
# - تنظیمات

# ═══════════════════════════════════════════════════════════
# پشتیبان‌گیری (Docker)
# ═══════════════════════════════════════════════════════════

# بکاپ دیتابیس
docker exec openproject backup

# یا به صورت دستی
docker exec openproject pg_dump -U postgres openproject > backup.sql

# بکاپ volumes
docker run --rm \
  -v openproject_data:/data \
  -v $(pwd):/backup \
  alpine tar czf /backup/openproject_data.tar.gz /data

docker run --rm \
  -v openproject_pgdata:/data \
  -v $(pwd):/backup \
  alpine tar czf /backup/openproject_pgdata.tar.gz /data

# ═══════════════════════════════════════════════════════════
# بازیابی (Packaged Installation)
# ═══════════════════════════════════════════════════════════

# لیست بکاپ‌ها
ls -la /var/db/openproject/backup/

# بازیابی
sudo openproject run restore /path/to/backup.tar.gz

# ═══════════════════════════════════════════════════════════
# بازیابی (Docker)
# ═══════════════════════════════════════════════════════════

# توقف کانتینر
docker-compose down

# بازیابی volumes
docker run --rm \
  -v openproject_data:/data \
  -v $(pwd):/backup \
  alpine sh -c "cd /data && tar xzf /backup/openproject_data.tar.gz --strip 1"

docker run --rm \
  -v openproject_pgdata:/data \
  -v $(pwd):/backup \
  alpine sh -c "cd /data && tar xzf /backup/openproject_pgdata.tar.gz --strip 1"

# شروع مجدد
docker-compose up -d

# ═══════════════════════════════════════════════════════════
# بکاپ خودکار با Cron
# ═══════════════════════════════════════════════════════════

# ویرایش crontab

CLI Commands 
# ═══════════════════════════════════════════════════════════
# Packaged Installation Commands
# ═══════════════════════════════════════════════════════════

# مدیریت سرویس
sudo openproject start
sudo openproject stop
sudo openproject restart
sudo openproject status

# پیکربندی
sudo openproject configure
sudo openproject reconfigure

# بکاپ و بازیابی
sudo openproject run backup
sudo openproject run restore /path/to/backup.tar.gz

# کنسول Rails
sudo openproject run console

# اجرای rake tasks
sudo openproject run rake db:migrate
sudo openproject run rake assets:precompile

# لاگ‌ها
sudo openproject logs
sudo openproject logs -f  # follow

# ═══════════════════════════════════════════════════════════
# Docker Commands
# ═══════════════════════════════════════════════════════════

# مدیریت کانتینر
docker-compose up -d
docker-compose down
docker-compose restart
docker-compose logs -f

# کنسول Rails
docker exec -it openproject bundle exec rails console

# بکاپ
docker exec openproject backup

# Shell access
docker exec -it openproject bash
