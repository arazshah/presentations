شروع سریع

دستورات اولیه برای راه‌اندازی پروژه

1️⃣ Clone و نصب Dependencies

# Clone repository
git clone https://github.com/yourusername/geochat-ai.git
cd geochat-ai

# Install dependencies
pnpm install

# Setup environment
cp .env.example .env.local

2️⃣ راه‌اندازی Database

# Start Docker containers
docker-compose up -d

# Verify ClickHouse
docker exec -it clickhouse clickhouse-client

# Load sample data
python scripts/load_sample_data.py

3️⃣ اجرای Development Server

# Start all services
pnpm dev

# Or separately:
pnpm --filter web dev      # Frontend: http://localhost:3000
pnpm --filter api dev      # Backend:  http://localhost:8000

4️⃣ تست اولین Query

# Test MCP endpoint
curl -X POST http://localhost:8000/api/mcp/parse \
  -H "Content-Type: application/json" \
  -d '{"query": "Show cafes in Tehran"}'

# Expected response:
{
  "intent": "spatial_query",
  "entity_type": "cafe",
  "location": "Tehran",
  "filters": []
}
